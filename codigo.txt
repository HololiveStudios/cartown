    const EventBus = new Phaser.Events.EventEmitter();

    const PlayerState = {
      coins: 1500, cash: 10, xp: 120, level: 1, garageValue: 12000, carColor: 0xef4444,
      addRewards(coinsGained, xpGained) {
        this.coins += coinsGained; this.xp += xpGained;
        const req = this.level * 300;
        if (this.xp >= req) {
          this.level++; this.xp -= req; this.cash += 2;
          EventBus.emit('SHOW_NOTIFICATION', { title: '🎉 SUBIU DE NÍVEL!', subtitle: `Você alcançou o Nível ${this.level} (+2 Cash)` });
        }
        EventBus.emit('PLAYER_STATE_UPDATED', this);
      }
    };

    class GarageScene extends Phaser.Scene {
      constructor() { super('GarageScene'); }
      create() {
        const w = this.scale.width, h = this.scale.height;
        this.add.rectangle(w / 2, h / 2, w, h, 0x1e293b);
        const g = this.add.graphics(); g.lineStyle(1, 0x334155, 0.5);
        for (let i = -w; i < w * 2; i += 60) { g.lineBetween(i, 0, i + h, h); g.lineBetween(i, h, i + h, 0); }

        this.workBay = this.add.container(w / 2 - 180, h / 2);
        const bayBase = this.add.rectangle(0, 0, 160, 100, 0x0f172a).setStrokeStyle(3, 0x3b82f6);
        this.add.text(w / 2 - 180, h / 2 - 35, 'ELEVADOR #1', { fontSize: '11px', fontStyle: 'bold', color: '#60a5fa' }).setOrigin(0.5);
        this.bayStatus = this.add.text(w / 2 - 180, h / 2, 'CLIQUE P/ SERVIÇO\n(Lavar Carro)', { fontSize: '10px', align: 'center', color: '#94a3b8' }).setOrigin(0.5);
        bayBase.setInteractive({ useHandCursor: true }).on('pointerdown', () => this.startWorkBayJob());

        this.carContainer = this.add.container(w / 2 + 100, h / 2);
        const shadow = this.add.ellipse(0, 25, 150, 40, 0x000000, 0.4);
        this.carBody = this.add.rectangle(0, 0, 140, 60, 0xffffff).setStrokeStyle(3, 0x0f172a);
        this.carBody.setTint(PlayerState.carColor);
        const w1 = this.add.circle(-45, 25, 12, 0x020617).setStrokeStyle(2, 0x94a3b8);
        const w2 = this.add.circle(45, 25, 12, 0x020617).setStrokeStyle(2, 0x94a3b8);
        const glass = this.add.rectangle(10, -10, 60, 25, 0x38bdf8, 0.6);
        this.carContainer.add([shadow, this.carBody, w1, w2, glass]);

        EventBus.on('CHANGE_CAR_COLOR', (color) => {
          PlayerState.carColor = color;
          this.carBody.setTint(color);
        });
        this.jobActive = false;
      }

      startWorkBayJob() {
        if (this.jobActive) return;
        this.jobActive = true;
        this.bayStatus.setText('SERVIÇO EM ANDAMENTO...\n(10s)').setColor('#eab308');
        this.time.delayedCall(10000, () => {
          this.jobActive = false;
          this.bayStatus.setText('✅ CONCLUÍDO!\n+150 Moedas | +30 XP').setColor('#22c55e');
          PlayerState.addRewards(150, 30);
        });
      }
    }

    class DragRaceScene extends Phaser.Scene {
      constructor() { super('DragRaceScene'); }
      create() {
        const w = this.scale.width, h = this.scale.height;
        this.add.rectangle(w / 2, h / 2, w, h, 0x090d16);
        this.add.rectangle(w / 2, h - 140, w, 180, 0x1e293b);
        this.add.rectangle(w - 120, h - 140, 20, 180, 0xffffff);

        const cx = w / 2;
        this.add.rectangle(cx, 160, 70, 200, 0x0f172a).setStrokeStyle(3, 0x334155);
        this.y1 = this.add.circle(cx, 90, 18, 0x334155);
        this.y2 = this.add.circle(cx, 135, 18, 0x334155);
        this.y3 = this.add.circle(cx, 180, 18, 0x334155);
        this.green = this.add.circle(cx, 225, 20, 0x334155);

        this.playerCar = this.add.container(120, h - 140);
        const body = this.add.rectangle(0, 0, 130, 50, 0xffffff).setStrokeStyle(3, 0x000000);
        body.setTint(PlayerState.carColor);
        const w1 = this.add.circle(-40, 22, 11, 0x000000);
        const w2 = this.add.circle(40, 22, 11, 0x000000);
        this.playerCar.add([body, w1, w2]);

        this.raceState = 'IDLE'; this.greenTime = 0; this.currentRPM = 1000; this.currentGear = 1;
        this.input.keyboard.on('keydown-SPACE', () => this.handleAction());
        EventBus.on('PLAYER_SHIFT_INPUT', () => this.handleAction());

        this.startRaceSequence();
      }

      startRaceSequence() {
        this.raceState = 'COUNTDOWN';
        EventBus.emit('SHOW_NOTIFICATION', { title: 'PREPARE-SE...', subtitle: 'Aguarde o sinal verde para arrancar' });
        this.time.delayedCall(600, () => { if (this.raceState==='COUNTDOWN') this.y1.setFillStyle(0xeab308); });
        this.time.delayedCall(1200, () => { if (this.raceState==='COUNTDOWN') this.y2.setFillStyle(0xeab308); });
        this.time.delayedCall(1800, () => { if (this.raceState==='COUNTDOWN') this.y3.setFillStyle(0xeab308); });
        this.time.delayedCall(2400, () => {
          if (this.raceState === 'COUNTDOWN') {
            this.raceState = 'GREEN'; this.green.setFillStyle(0x22c55e); this.greenTime = this.time.now;
            EventBus.emit('SHOW_NOTIFICATION', { title: '🚦 ACELERE! (GO!)', subtitle: 'Pressione Espaço para arrancar' });
          }
        });
      }

      handleAction() {
        if (this.raceState === 'COUNTDOWN') {
          this.raceState = 'FOUL';
          this.y1.setFillStyle(0xef4444); this.y2.setFillStyle(0xef4444); this.y3.setFillStyle(0xef4444);
          EventBus.emit('SHOW_NOTIFICATION', { title: '❌ QUEIMOU A LARGADA!', subtitle: 'Você acelerou antes do sinal verde' });
          return;
        }
        if (this.raceState === 'GREEN') {
          this.raceState = 'RACING';
          const rx = ((this.time.now - this.greenTime) / 1000).toFixed(3);
          EventBus.emit('SHOW_NOTIFICATION', { title: `🟢 LARGADA! (${rx}s)`, subtitle: 'Troque de marcha na zona verde' });
          document.getElementById('rpm-container').classList.remove('hidden');
          return;
        }
        if (this.raceState === 'RACING') this.shiftGear();
      }

      shiftGear() {
        if (this.currentGear >= 4) return;
        const rpm = this.currentRPM;
        let q = 'normal', boost = 1.0, msg = 'TROCA NORMAL';
        if (rpm >= 6500 && rpm <= 7500) { q = 'perfect'; boost = 1.35; msg = '⚡ PERFECT SHIFT!'; }
        else if (rpm < 6500) { msg = '⚠️ TROCOU CEDO'; boost = 0.8; }
        else { msg = '❌ TROCA TARDIA'; boost = 0.7; }

        this.currentGear++; this.currentRPM = 3800; this.playerCar.x += 60 * boost;
        EventBus.emit('SHIFT_FEEDBACK', { gear: this.currentGear, feedback: msg, quality: q });
        if (this.currentGear === 4 || this.playerCar.x > this.scale.width - 200) this.finishRace();
      }

      update(time, delta) {
        if (this.raceState !== 'RACING') return;
        this.currentRPM += (30 / this.currentGear);
        if (this.currentRPM >= 8000) this.currentRPM = 7800;
        this.playerCar.x += (this.currentGear * 60 + this.currentRPM / 80) * (delta / 1000);
        EventBus.emit('UPDATE_RPM_UI', { rpm: Math.floor(this.currentRPM), gear: this.currentGear });
        if (this.playerCar.x >= this.scale.width - 150) this.finishRace();
      }

      finishRace() {
        if (this.raceState === 'FINISHED') return;
        this.raceState = 'FINISHED';
        document.getElementById('rpm-container').classList.add('hidden');
        EventBus.emit('SHOW_NOTIFICATION', { title: '🏆 CORRIDA CONCLUÍDA!', subtitle: '+200 Moedas | +50 XP Recompensas' });
        PlayerState.addRewards(200, 50);
      }
    }

    const config = {
      type: Phaser.AUTO, parent: 'game-container',
      width: window.innerWidth, height: window.innerHeight,
      scale: { mode: Phaser.Scale.RESIZE, autoCenter: Phaser.Scale.CENTER_BOTH },
      backgroundColor: '#020617', scene: [GarageScene, DragRaceScene]
    };

    const game = new Phaser.Game(config);

    EventBus.on('PLAYER_STATE_UPDATED', (state) => {
      document.getElementById('coins-display').innerText = state.coins.toLocaleString();
      document.getElementById('cash-display').innerText = state.cash;
      document.getElementById('gv-display').innerText = `$${state.garageValue.toLocaleString()}`;
      document.getElementById('level-display').innerText = state.level;
      document.getElementById('level-badge').innerText = state.level;
      document.getElementById('xp-bar').style.width = `${Math.min(100, (state.xp / (state.level * 300)) * 100)}%`;
    });

    EventBus.on('SHOW_NOTIFICATION', (data) => {
      document.getElementById('notif-title').innerText = data.title;
      document.getElementById('notif-subtitle').innerText = data.subtitle;
      const b = document.getElementById('notification-banner');
      b.classList.remove('hidden');
      setTimeout(() => b.classList.add('hidden'), 3500);
    });

    EventBus.on('UPDATE_RPM_UI', (data) => {
      document.getElementById('rpm-bar').style.width = `${(data.rpm / 8000) * 100}%`;
      document.getElementById('rpm-text').innerText = data.rpm;
      document.getElementById('gear-display').innerText = data.gear;
    });

    EventBus.on('SHIFT_FEEDBACK', (data) => {
      const f = document.getElementById('shift-feedback');
      f.innerText = data.feedback;
      f.className = data.quality === 'perfect' 
        ? "text-xs font-black text-green-400 bg-green-950 px-2.5 py-1 rounded-lg border border-green-500/30"
        : "text-xs font-black text-amber-400 bg-amber-950 px-2.5 py-1 rounded-lg";
    });

    let currentScene = 'GarageScene';
    document.getElementById('btn-toggle-scene').addEventListener('click', () => {
      if (currentScene === 'GarageScene') {
        currentScene = 'DragRaceScene';
        game.scene.stop('GarageScene'); game.scene.start('DragRaceScene');
        document.getElementById('btn-toggle-scene').innerText = '🏠 Voltar para Garagem';
      } else {
        currentScene = 'GarageScene';
        game.scene.stop('DragRaceScene'); game.scene.start('GarageScene');
        document.getElementById('btn-toggle-scene').innerText = '🚦 Ir Correr (Arrancada)';
        document.getElementById('rpm-container').classList.add('hidden');
      }
    });

    const modal = document.getElementById('customizer-modal');
    document.getElementById('btn-open-customizer').addEventListener('click', () => modal.classList.remove('hidden'));
    document.getElementById('btn-close-modal').addEventListener('click', () => modal.classList.add('hidden'));

    window.setCarColor = function(color) { EventBus.emit('CHANGE_CAR_COLOR', color); };
    document.getElementById('btn-shift-input').addEventListener('click', () => { EventBus.emit('PLAYER_SHIFT_INPUT'); });
  </script>
</body>
